```bash
git log --oneline -1        # docs 커밋
npm install
npx next typegen            # ★ 안 돌리면 다음 줄이 3건 뜬다
npx tsc --noEmit            # 0건
npm run lint                # 깨끗
npm run gen:seed            # 아무 파일도 안 바뀌어야 한다
npm run test:scoring        # 0 실패. 통과 수는 감시가 늘 때마다 는다 — 여기 안 적는다
                            # ★ 수를 적으면 커밋마다 낡는다. 실제로 1597 에 멈춰 있었다
npm run check:numbers       # 낡은 수 0건 — 주석·md 는 test:scoring 이 안 본다
npm run ai:probe -- --dry   # 자격 없이 돈다. 표본·프롬프트·틀 길이
                            # ★ DB 자격이 있으면 --check 로 마개까지 재라
npx tsx scripts/at-table.ts # 10단계 배정표
```
